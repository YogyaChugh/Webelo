191# Position based on screen size
192def compute_positions(node, depth=0, x=0):
193 if "children" not in node or not node["children"]:
194 node["x"] = x
195 node["y"] = depth * (NODE_HEIGHT + Y_SPACING)
196 return x + NODE_WIDTH + X_SPACING
197 child_x = x
198 for child in node["children"]:
199 child_x = compute_positions(child, depth + 1, child_x)
200 left = node["children"][0]["x"]
201 right = node["children"][-1]["x"]
202 node["x"] = (left + right) // 2
203 node["y"] = depth * (NODE_HEIGHT + Y_SPACING)
204
205 return child_x

286def print_children(some,logs):
287 logs = logs
288 dom_tree = {}
289 if (type(some)!=bs4.element.NavigableString):
290 if some.name:
291 dom_tree["name"] = some.name
292 dom_tree["attrs"] = some.attrs
293 dom_tree["content"] = ""
294 if some.string:
295 dom_tree['content'] = some.string
296 dom_tree['children'] = []
297 logs+=f"Created Element {some.name}\n"
298 try:
299 d = 0
300 for child in some.children:
301 if d==0: dom_tree["children"] = []
302 d+=1
303 pp,logs = print_children(child,logs)
304 if pp!={}:
305 dom_tree["children"].append(pp)
306 except:
307 pass
308 else:
309 dom_tree["name"] = "Text"
310 dom_tree['attrs'] = {}
311 dom_tree["content"] = ""
312 if some.string:
313 dom_tree['content'] = some.string
314 dom_tree['children'] = []
315 logs+=f"Created Text Node\n"
316 return dom_tree, logs
